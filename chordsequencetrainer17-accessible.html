// --- Accessibility: screen reader announcements --- let liveTimer; function announce(text, urgent = false) { const container = document.getElementById('srLive'); const focusEl = document.getElementById('srFocus'); if (!container) return; container.innerHTML = ''; clearTimeout(liveTimer); liveTimer = setTimeout(() => { // Approach 1: dynamically insert a role=alert node (works on some iOS builds) const alertEl = document.createElement('div'); alertEl.setAttribute('role', urgent ? 'alert' : 'status'); alertEl.textContent = text; container.appendChild(alertEl); // Approach 2: focus a hidden element with this text as aria-label // (this is the reliable one on iOS WKWebView) if (focusEl) { focusEl.setAttribute('aria-label', text); focusEl.textContent = text; try { focusEl.focus({ preventScroll: true }); } catch (e) {} } // Cleanup old nodes so the DOM doesn't grow indefinitely setTimeout(() => { while (container.firstChild) container.removeChild(container.firstChild); }, 9000); }, 60); }
