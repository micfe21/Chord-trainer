function populateMidiDevices(){ if (!midiAccess) { midiDeviceSelect.innerHTML = ''; return; } let inputs = []; let outputs = []; try { // Use Array.from instead of spread, and check for existence if (midiAccess.inputs && typeof midiAccess.inputs.values === 'function') { inputs = Array.from(midiAccess.inputs.values()); } if (midiAccess.outputs && typeof midiAccess.outputs.values === 'function') { outputs = Array.from(midiAccess.outputs.values()); } } catch (e) { console.warn('Error reading MIDI devices:', e); inputs = []; outputs = []; } const parts = []; if (inputs.length) { parts.push('' + inputs.map(input => { const label = `${input.name || 'Input'}${input.manufacturer ? ' — ' + input.manufacturer : ''}`; return ``; }).join('') + ''); } if (outputs.length) { parts.push('' + outputs.map(output => { const label = `${output.name || 'Output'}${output.manufacturer ? ' — ' + output.manufacturer : ''}`; return ``; }).join('') + ''); } midiDeviceSelect.innerHTML = parts.join('') || ''; if (!selectedMidiInputId && inputs[0]) selectedMidiInputId = inputs[0].id; if (!selectedMidiOutputId && outputs[0]) selectedMidiOutputId = outputs[0].id; const preferred = selectedMidiInputId ? `in:${selectedMidiInputId}` : (selectedMidiOutputId ? `out:${selectedMidiOutputId}` : ''); if (preferred) midiDeviceSelect.value = preferred; bindMidiInputs(); const connected = inputs.length > 0 || outputs.length > 0; setMidiStatus(connected ? 'ready' : 'no devices', connected); }
