const inputVoices = new Map(); function startInputNote(note, velocity = 96) { const ctx = ensureAudio(); // If the same note is already sounding, stop it first const existing = inputVoices.get(note); if (existing) { try { existing.osc.stop(); } catch {} inputVoices.delete(note); } const osc = ctx.createOscillator(); const gain = ctx.createGain(); osc.type = 'triangle'; osc.frequency.value = 440 * Math.pow(2, (note - 69) / 12); const amp = Math.max(0.02, Math.min(0.25, (velocity / 127) * 0.25)); gain.gain.setValueAtTime(0.0001, ctx.currentTime); gain.gain.exponentialRampToValueAtTime(amp, ctx.currentTime + 0.02); osc.connect(gain); gain.connect(masterGain || ctx.destination); osc.start(); inputVoices.set(note, { osc, gain }); } function stopInputNote(note) { const voice = inputVoices.get(note); if (!voice) return; const ctx = ensureAudio(); voice.gain.gain.cancelScheduledValues(ctx.currentTime); voice.gain.gain.setTargetAtTime(0.0001, ctx.currentTime, 0.03); try { voice.osc.stop(ctx.currentTime + 0.12); } catch {} inputVoices.delete(note); } function clearInputVoices() { for (const note of inputVoices.keys()) { stopInputNote(note); } inputVoices.clear(); }
